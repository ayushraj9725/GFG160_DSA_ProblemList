Problem : Minimumm Path Sum 

Problem Statement : 

Given a m x n grid filled with non-negative numbers, find a path from top left to bottom right, which minimizes the sum of all numbers along its path.

Note: You can only move either down or right at any point in time.

Example 1:
Input: grid = [[1,3,1],[1,5,1],[4,2,1]]
Output: 7
Explanation: Because the path 1 → 3 → 1 → 1 → 1 minimizes the sum.

Example 2:
Input: grid = [[1,2,3],[4,5,6]]
Output: 12

Constraints:
m == grid.length
n == grid[i].length
1 <= m, n <= 200
0 <= grid[i][j] <= 200


/************************************ PROBLEM SOLUTION ************************************/

/// RECURSIVE APPROACH ///  Java  : TIME : O(2^n) and Space : O(0)

class Solution {
    public int pathSum(int i ,int j, int m ,int n, int[][] grid){
        // base case
        if(i == m-1 && j == n-1){
            return grid[i][j]; 
        }

        if(i >= m || j >= n){
            return Integer.MAX_VALUE ; // out of bound phase
        }

        int right = pathSum(i+1,j,m,n,grid);
        int left = pathSum(i,j+1,m,n,grid);

        return grid[i][j] + Math.min(right,left); // take the minimest value and add into current 
    }
    public int minPathSum(int[][] grid) {
        
        int m = grid.length;
        int n = grid[0].length ;

        return pathSum(0,0,m,n,grid);
        
    }
}

/// TOP DOWN APPROACH or MEMORIZATION /// Java   : TIMEC OMP : O(N+M) and Space : O(N+M) ;

class Solution {
    public int pathSum(int i ,int j, int m ,int n, int[][] grid, int[][] memo){

        if(i >= m || j >= n){
            return Integer.MAX_VALUE ; // out of bound phase
        }
        
        // base case
        if(i == m-1 && j == n-1){
            return grid[i][j]; 
        }
        
        // check if path already visited or not
        if(memo[i][j] != -1){
            return memo[i][j] ;
        }

        int right = pathSum(i+1,j,m,n,grid,memo);
        int left = pathSum(i,j+1,m,n,grid,memo);

        memo[i][j] = grid[i][j] + Math.min(right,left);
        return memo[i][j] ; // take the minimest value and add into current 

    }
    public int minPathSum(int[][] grid) {
        
        int m = grid.length;
        int n = grid[0].length ;

        int memo[][] = new int[m][n];

        for(int[] arr : memo){
            Arrays.fill(arr,-1);
        }

        return pathSum(0,0,m,n,grid,memo);
        
    }
}

/// BOTTOM UP APPROACH or TABULATION /// JAVA : TIME COMP : O(N+M) and Space : O(N+M) ;

class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        
        int m = grid.size();
        int n = grid[0].size();

        vector<vector<int>> dp(m,vector<int>(n));

        // base case
        dp[m-1][n-1] = grid[m-1][n-1] ;

        // fill the right most cell (last column)
        for(int i = m-2 ; i >= 0 ; i--){
            dp[i][n-1] = grid[i][n-1] + dp[i+1][n-1];
        }

        // fill the bottom most cell (last row)
        for(int j = n-2 ; j >= 0 ; j--){
            dp[m-1][j] = grid[m-1][j] + dp[m-1][j+1];
        }

        // now fill the remaining cell in between

        for(int i = m-2 ; i >= 0 ; i--){
            for(int j = n-2 ; j >= 0 ; j--){
                dp[i][j] = grid[i][j] + min(dp[i+1][j],dp[i][j+1]);
            }
        }

        return dp[0][0] ; // answer we have stored the minimest path sum 
    }
};
