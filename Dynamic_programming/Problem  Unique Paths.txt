Problem : Unique Paths

Problem Statement : 
There is a robot on an m x n grid. The robot is initially located at the top-left corner (i.e., grid[0][0]). The robot tries to move to the bottom-right corner (i.e., grid[m - 1][n - 1]). The robot can only move either down or right at any point in time.

Given the two integers m and n, return the number of possible unique paths that the robot can take to reach the bottom-right corner.
The test cases are generated so that the answer will be less than or equal to 2 * 109.

Example 1:
Input: m = 3, n = 7
Output: 28

Example 2:
Input: m = 3, n = 2
Output: 3
Explanation: From the top-left corner, there are a total of 3 ways to reach the bottom-right corner:

1. Right -> Down -> Down
2. Down -> Down -> Right
3. Down -> Right -> Down

Constraints:
1 <= m, n <= 100


/**********************************************  PROBLEM SOLUTION *******************************************/

// RECURSIVE APPROACH //   C++ 

class Solution {
public:
    int uniquePaths(int m, int n) {
        
          return traverseAllPath(0,0,m,n);

    }

    int traverseAllPath(int i,int j ,int m,int n){
        if(i == m-1 || j == n-1){
            return 1 ;
        }


        int path1 = 0 ;
        int path2 = 0 ;
        if(i < m-1){
            path1 = traverseAllPath(i+1,j,m,n);
        }

        if(j < n-1){
            path2 = traverseAllPath(i,j+1,m,n);
        }

        return path1 + path2 ;
    }
};

/// TOP DOWN APPROACH //    ********* C++ **********

class Solution {
public:
    int uniquePaths(int m, int n) {
        
        vector<vector<int>> memo(m,vector<int>(n,-1)) ;
        return traverseAllPath(0,0,m,n,memo);

    }

    int traverseAllPath(int i,int j ,int m,int n,vector<vector<int>> memo){
        if(i == m-1 || j == n-1){
            return 1 ;
        }

        // check if already visit the path

        if(memo[i][j] != -1){
           return memo[i][j];
        }

        int path1 = 0 ;
        int path2 = 0 ;
        if(i < m-1){
            path1 = traverseAllPath(i+1,j,m,n,memo);
        }

        if(j < n-1){
            path2 = traverseAllPath(i,j+1,m,n,memo);
        }

        memo[i][j] = path1 + path2 ;

        return memo[i][j];
    }
};

//// BOTTOM UP APPROACH /// ********** JAVA **********


class Solution {
    public int uniquePaths(int m, int n) {
        
        int[][] dp = new int[m][n] ;

        for(int i = 0 ;i < m ; i++){
            dp[i][n-1] = 1;
        }
        for(int j = 0 ; j < n ; j++){
            dp[m-1][j] = 1 ;
        }

        for(int i = m - 2 ; i >= 0 ; i--){
            for(int j = n - 2 ;j >= 0 ;j--){
                dp[i][j] = dp[i+1][j] + dp[i][j+1];
            }
        }

        return dp[0][0] ;
    }
}
